#!/bin/bash
set -x

# setup eth1 - limit bitrate and buffer
sudo tc qdisc del dev eth1 root
sudo tc qdisc add dev eth1 root handle 1: htb default 3
sudo tc class add dev eth1 parent 1: classid 1:3 htb rate 0.1Mbit
sudo tc qdisc add dev eth1 parent 1:3 handle 3: bfifo limit 0.01MB

# setup eth2 - limit bitrate and buffer
sudo tc qdisc del dev eth2 root
sudo tc qdisc add dev eth2 root handle 1: htb default 3
sudo tc class add dev eth2 parent 1: classid 1:3 htb rate 0.1Mbit
sudo tc qdisc add dev eth2 parent 1:3 handle 3: bfifo limit 0.01MB
